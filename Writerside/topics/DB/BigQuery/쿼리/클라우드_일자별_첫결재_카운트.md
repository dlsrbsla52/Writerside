# 클라우드 일자별 첫결재 카운트

```sql
with temp as (select DATE(PAY_DATE) as PAY_DATE, USER_ID
              from data.TB_KPI_RAW_SALE
              where MEM_CLASS = '선생님'
                and PRODUCT_CLASS like '%클라우드%'
                and PAY_DATE >= '2024-08-21 00:00:00.000000'
                and PAY_DATE < '2024-09-06 00:00:00.000000'
                and FIRST_PAY_GUBUN != '환불'
                and PAY_TYPE != '전체취소'
              group by DATE(PAY_DATE), USER_ID)
, fpay_user AS
(select t.REG_DT_YMD, t.CID, t.USER_ID
from `data.TB_TRACKING_LOG*` as t
         join temp on date(t.REG_DT) = temp.PAY_DATE and t.USER_ID = temp.USER_ID
where REG_DT >= '2024-08-21 00:00:00.000000'
  and REG_DT < '2024-09-06 00:00:00.000000'
  and EVENT_ACTION_TARGET_CD = 'EX8KTFPYAO48MWAHEVOBFOPZE20240813'
group by t.CID, t.REG_DT_YMD, t.USER_ID
)
SELECT REG_DT_YMD, COUNT(REG_DT_YMD)
FROM fpay_user
GROUP BY REG_DT_YMD
ORDER BY REG_DT_YMD
```
