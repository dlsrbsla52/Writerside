# 클라우드 유사문항 클릭 트레킹 쿼리

```sql
select REG_DT,  
       CID,
       CASE  
           WHEN EVENT_ACTION_TARGET_CD = '6QA47N16D1D5KJ3C1J6ER9MD120240829' THEN 'AI유사문제버튼클릭'  
           WHEN EVENT_ACTION_TARGET_CD = 'HWLJ5R4IAHTAZXXMHIULLU3CQ20240829' THEN '문항추가'  
           WHEN EVENT_ACTION_TARGET_CD = 'NUG9QBFV6J17CIN7AL86C9TW920240829' THEN '교체버튼클릭'  
           ELSE 'ETC'  
       END               AS EVENT_ACTION_TARGET_NAME  
       ,  
       PAGE_URL,  
       REG_DT_YMD  
from `data.TB_TRACKING_LOG*`  
where REG_DT >= '2024-08-30 00:00:00.000000'  
  and REG_DT <= '2024-09-06 00:00:00.000000'  
  and EVENT_ACTION_TARGET_CD in ('6QA47N16D1D5KJ3C1J6ER9MD120240829', 'HWLJ5R4IAHTAZXXMHIULLU3CQ20240829', 'NUG9QBFV6J17CIN7AL86C9TW920240829')  
  and LENGTH(CID) = 32  
group by REG_DT_YMD, CID, EVENT_ACTION_TARGET_CD, PAGE_URL, REG_DT  
order by REG_DT_YMD, PAGE_URL
```
