# 빅쿼리 비용 카운트

## 날짜별 쿼리 비용
```sql
SELECT DATE(TIMESTAMP(usage_start_time), 'Asia/Seoul') AS `날짜`,  
       service.description                             AS `Service Description`,  
       SUM(CAST(cost AS NUMERIC))                      AS `비용`,  
       SUM(IFNULL((SELECT SUM(CAST(c.amount AS numeric))  
                   FROM UNNEST(credits) c  
                   WHERE c.type IN ('')), 0))          AS `할인`,  
       SUM(IFNULL((SELECT SUM(CAST(c.amount AS numeric))  
                   FROM UNNEST(credits) c  
                   WHERE c.type IN ('')), 0))          AS `프로모션 및 기타`,  
       SUM(CAST(cost AS NUMERIC)) + SUM(IFNULL((SELECT SUM(CAST(c.amount AS numeric))  
                                                FROM UNNEST(credits) c  
                                                WHERE c.type IN ('')), 0)) +  
       SUM(IFNULL((SELECT SUM(CAST(c.amount AS numeric))  
                   FROM UNNEST(credits) c  
                   WHERE c.type IN ('')), 0))          AS `소계`  
FROM `zocbodb.data.gcp_billing_export_resource_v1_010C3D_D6C6C5_D4DF78`  
WHERE cost_type != 'tax'  
  AND cost_type != 'adjustment'  
  AND service.description = 'BigQuery'  -- BigQuery 서비스 비용만 필터링  
  AND TIMESTAMP(usage_start_time) BETWEEN  
    TIMESTAMP('2024-09-01 00:00:00', 'Asia/Seoul')  
    AND TIMESTAMP('2024-09-30 00:00:00', 'Asia/Seoul')  
GROUP BY `날짜`,  
         service.description  
ORDER BY `날짜` ASC,  
         `Service Description` ASC;
```


## 날짜별 쿼리내역(사람,시간) 및 쿼리당 비용
```sql
SELECT user_email,  
       query,  
       total_bytes_billed / 1e9 AS gigabytes_billed,  
       total_bytes_billed / 1e9 * 5 AS estimated_cost,  
       FORMAT_TIMESTAMP("%Y-%m-%d %H:%M:%S", TIMESTAMP_ADD(creation_time, INTERVAL 9 HOUR)) AS creation_time_seoul  
FROM `region-asia-northeast3`.INFORMATION_SCHEMA.JOBS_BY_PROJECT  
WHERE TIMESTAMP_ADD(creation_time, INTERVAL 9 HOUR) >= '2024-09-19 00:00:00'  
  AND TIMESTAMP_ADD(creation_time, INTERVAL 9 HOUR) < '2024-09-20 00:00:00'  
ORDER BY gigabytes_billed DESC;
```


```sql
SELECT  
    user_email,  
    query,  
    total_bytes_processed/1e9 AS gigabytes_processed,  
    FORMAT_TIMESTAMP("%Y-%m-%d %H:%M:%S", TIMESTAMP_ADD(creation_time, INTERVAL 9 HOUR)) AS creation_time_seoul,  
    total_bytes_billed/1e9 AS gigabytes_billed  
FROM  
    `region-asia-northeast3`.INFORMATION_SCHEMA.JOBS_BY_PROJECT  
WHERE creation_time >= '2024-11-01 00:00:00'  
  AND creation_time < '2024-12-30 00:00:00'  
  AND job_type = "QUERY"  
  AND state = "DONE"  
  AND user_email = 'hdpark772@eduzone.co.kr'  
ORDER BY  
    gigabytes_billed, creation_time_seoul DESC;
```


## 사용자별 쿼리 금액
```sql
SELECT user_email,  
       SUM(total_bytes_billed / 1e9 * 5) AS total_estimated_cost,  
       COUNT(query) AS query_count,  -- Optional: Number of queries executed by the user  
       FORMAT_TIMESTAMP("%Y-%m-%d", TIMESTAMP_ADD(MIN(creation_time), INTERVAL 9 HOUR)) AS first_query_seoul,  -- First query time in the range  
       FORMAT_TIMESTAMP("%Y-%m-%d", TIMESTAMP_ADD(MAX(creation_time), INTERVAL 9 HOUR)) AS last_query_seoul   -- Last query time in the range  
FROM `region-asia-northeast3`.INFORMATION_SCHEMA.JOBS_BY_PROJECT  
WHERE TIMESTAMP_ADD(creation_time, INTERVAL 9 HOUR) >= '2024-09-24 00:00:00'  
  AND TIMESTAMP_ADD(creation_time, INTERVAL 9 HOUR) < '2024-09-25 00:00:00'  
GROUP BY user_email  
ORDER BY total_estimated_cost DESC;
```
